Answer 5:
#include <stdio.h>

void swapTwoNumbers(int *a, int *b)
{
    int temp;

    temp = *a;
    *a = *b;
    *b = temp;
}

int main()
{
    int a = 10;
    int b = 20;

    printf("Before a: %d\n", a);
    printf("Before b: %d\n", b);

    swapTwoNumbers(&a, &b);

    printf("After a: %d\n", a);
    printf("After b: %d\n", b);

    return 0;
}

#include <stdio.h>

int addTwoNumbers(int *a, int *b)
{
    int result;
    result = (&a) + (&b);
    return result;
}

int main()
{
    int a = 10, b = 20;
    printf("%d + %d = %d\n", a, b, addTwoNumbers(a, b));

    return 0;
}
